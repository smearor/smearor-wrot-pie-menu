fn set_activation_threshold(&self, threshold: f64)

Sets the pinch-to-zoom activation threshold. The pie menu opens when the zoom scale exceeds this value. Default: 3.5.

Source

fn activation_threshold(&self) -> f64

Returns the current activation threshold.

Source

fn set_deactivation_threshold(&self, threshold: f64)

Sets the pinch-out deactivation threshold. The pie menu closes when the zoom scale drops below this value. Default: 0.5.

Source

fn deactivation_threshold(&self) -> f64

Returns the current deactivation threshold.

Source

fn set_rotation_gesture_enabled(&self, enabled: bool)

Enables or disables the rotation gesture when the pie menu is open. When disabled, the rotation gesture controller is set to PropagationPhase::None. Default: true.

Source

fn rotation_gesture_enabled(&self) -> bool

Returns whether the rotation gesture is currently enabled.

Source

fn set_markings_enabled(&self, enabled: bool)

Enables or disables drawing of inner and outer ring markings. Default: true.

Source

fn markings_enabled(&self) -> bool

Returns whether ring markings are currently enabled.

Source

fn set_scroll_rotation_step(&self, sensitivity: f64)

Sets the scroll rotation sensitivity multiplier. The rotation delta is computed as dy * sensitivity, so higher values produce faster rotation. Default: 5.0.

Source

fn scroll_rotation_step(&self) -> f32

Returns the current scroll rotation sensitivity multiplier.

Source

fn cycle_selection(&self, direction: i32)

Cycles the keyboard selection by direction (-1 for CCW, +1 for CW). Items are sorted by angle to ensure deterministic navigation order.

Source

fn select_first_item(&self)

Selects the first item (smallest angle, typically 0°).

Source

fn confirm_selection(&self)

Confirms the current keyboard selection by sending PieMenuMessage::Event for the selected item. Does nothing if no item is selected or if the selected item is disabled.

Source

fn handle_left_stick_x(&self, x: f32)

Updates the stored left stick X-axis value for continuous rotation. The actual rotation is applied per-frame via the tick callback. Must be called on the GTK main thread. x is in the range [-1.0, 1.0].

Source

fn handle_right_stick(&self, x: f32, y: f32)

Selects the nearest menu item based on the right stick direction. x and y are in the range [-1.0, 1.0]. Must be called on the GTK main thread.

Source

fn find_nearest_item(&self, target_angle: f32) -> Option<String>

Finds the menu item whose angle is closest to target_angle (in degrees). Returns the item ID, or None if the menu is empty.

Source

fn set_keyboard_selection(&self, id: String)

Sets the keyboard selection to the given item ID and triggers a redraw.

Source

fn open_submenu(&self, parent_id: &str) -> Result<(), SubmenuError>

Opens the submenu of the item with the given id. Submenu item angles are redistributed (fixed items keep their angles, flexible items are auto-distributed) before rendering.

Source

fn close_submenu(&self) -> Result<(), SubmenuError>

Closes the current submenu and returns to the parent ring.

Source

fn submenu_depth(&self) -> u32

Returns the current submenu depth (0 = main ring).

Source

fn get_submenu_items(&self, parent_id: &str) -> Vec<MenuItem>

Returns the submenu items of the item with the given parent id.

Source

fn redistribute_submenu(&self, parent_id: &str)

Redistributes submenu item angles for the submenu of the item with the given parent id. Fixed items keep their angles; flexible items are re-spaced proportionally in the gaps.

Source

fn set_submenu_items( &self, parent_id: &str, items: Vec<MenuItem>, ) -> Result<(), SubmenuError>

Updates the submenu items of the item with the given parent id. Replaces the entire submenu item list. Triggers redistribution and overlap validation.

Source

fn set_pie_menu_radius(&self, radius: f32)

Sets the radius of the main pie menu ring in pixels. Default: 160.0.

Source

fn pie_menu_radius(&self) -> f32

Returns the current main pie menu ring radius in pixels.

Source

fn set_pie_menu_center_radius(&self, radius: f32)

Sets the inner radius of the pie menu ring (the transparent center) in pixels. Default: 64.0.

Source

fn pie_menu_center_radius(&self) -> f32

Returns the current inner radius of the pie menu ring in pixels.

Source

fn set_submenu_radius(&self, level: u32, radius: f32)

Sets the radius for a specific submenu level. Level 0 is the main ring, level 1 is the first submenu, etc.

Source

fn set_submenu_radius_step(&self, step: f32)

Sets the global step width between consecutive ring levels. Each submenu level’s radius is computed as: main_radius + level * step. Default: 80.0.

Source

fn max_submenu_depth(&self) -> u32

Returns the maximum submenu nesting depth.
