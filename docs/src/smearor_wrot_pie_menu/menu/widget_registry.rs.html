5/// Registry mapping widget type names to their factory implementations.
6///
7/// The registry is populated with standard implementations (`"circle"`,
8/// `"square"`) by the library. Consumers register custom widget
9/// factories via [`MenuItemWidgetRegistry::register`].
10///
11/// The registry is `!Send` and `!Sync` because factories produce
12/// `gtk4::Widget` instances bound to the GLib main thread.
13pub struct MenuItemWidgetRegistry {
14    factories: HashMap<String, Box<dyn MenuItemWidgetFactoryErased>>,
15}
16
17impl MenuItemWidgetRegistry {
18    /// Creates a new registry pre-populated with standard implementations.
19    ///
20    /// Standard implementations:
21    /// - `"circle"` - circular item with icon + label (existing behavior)
22    /// - `"square"` - square item with icon + label
23    /// - `"button"` - simple GTK4 Button with label (debug widget)
24    /// - `"gauge"` - tachometer-style gauge with color-coded zones
25    pub fn new() -> Self {
26        let mut registry = Self { factories: HashMap::new() };
27        registry.register(Box::new(crate::menu::circle_widget::CircleWidgetFactory));
28        registry.register(Box::new(crate::menu::square_widget::SquareWidgetFactory));
29        registry.register(Box::new(crate::menu::button_widget::ButtonWidgetFactory));
30        registry.register(Box::new(crate::menu::gauge_widget::GaugeWidgetFactory));
31        registry
32    }
33
34    /// Registers a custom widget factory under its type name.
35    ///
36    /// If a factory with the same type name already exists, it is
37    /// replaced. This allows consumers to override standard
38    /// implementations if desired.
39    pub fn register(&mut self, factory: Box<dyn MenuItemWidgetFactoryErased>) {
40        self.factories.insert(factory.type_name().to_string(), factory);
41    }
42
43    /// Resolves a factory by type name.
44    ///
45    /// Returns `None` if no factory is registered under the given name.
46    pub fn get(&self, type_name: &str) -> Option<&dyn MenuItemWidgetFactoryErased> {
47        self.factories.get(type_name).map(|boxed| boxed.as_ref())
48    }
49}
50
51impl Default for MenuItemWidgetRegistry {
52    fn default() -> Self {
53        Self::new()
54    }
55}

75    fn test_registry_register_custom() {
76        let mut registry = MenuItemWidgetRegistry::new();
77
78        struct CustomFactory;
79        impl MenuItemWidgetFactoryErased for CustomFactory {
80            fn type_name(&self) -> &str {
81                "custom"
82            }
83            fn build(&self, _item: &crate::menu::MenuItem, _context: &crate::menu::context::MenuItemContext) -> gtk4::Widget {
84                gtk4::Label::new(Some("custom")).upcast::<gtk4::Widget>()
85            }
86        }
87
88        registry.register(Box::new(CustomFactory));
89        assert!(registry.get("custom").is_some());
90    }

93    #[ignore = "requires GTK display environment"]
94    fn test_registry_register_overrides_existing() {
95        crate::test_util::ensure_gtk_init();
96        let mut registry = MenuItemWidgetRegistry::new();
97
98        struct ReplacementCircleFactory;
99        impl MenuItemWidgetFactoryErased for ReplacementCircleFactory {
100            fn type_name(&self) -> &str {
101                "circle"
102            }
103            fn build(&self, _item: &crate::menu::MenuItem, _context: &crate::menu::context::MenuItemContext) -> gtk4::Widget {
104                gtk4::Label::new(Some("replacement")).upcast::<gtk4::Widget>()
105            }
106        }
107
108        registry.register(Box::new(ReplacementCircleFactory));
109        let factory = registry.get("circle").expect("circle factory should exist");
110        let item = crate::menu::MenuItem::builder().id("test").angle(0.0).event("event").build();
111        let context = crate::menu::context::MenuItemContext {
112            id: "test".to_string(),
113            event: "event".to_string(),
114            trigger_event: Box::new(|| {}),
115        };
116        let widget = factory.build(&item, &context);
117        let label = widget.downcast_ref::<gtk4::Label>().expect("expected Label");
118        assert_eq!(label.text(), "replacement");
119    }
