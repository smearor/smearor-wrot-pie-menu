71    impl WidgetImpl for SquareItemWidgetImpl {
72        fn snapshot(&self, snapshot: &Snapshot) {
73            let obj = self.obj();
74            let center = obj.center();
75            let item_radius = Pixels::new(f64::from(self.item_radius.borrow().as_raw()));
76            let corner_radius = Pixels::new(8.0).min(item_radius * 0.2).as_f64() as f32;
77
78            // Centered square of the given edge length - converts to
79            // graphene coordinates at the draw boundary.
80            let bounds_of_size = |size: Pixels<f64>| graphene::Rect::from(Rect::<Pixels>::centered(center, Vec2::<Pixels>::splat(size)).to_f32());
81
82            let bg_color = *self.bg_color.borrow();
83            let disabled_alpha = if *self.enabled.borrow() { Alpha::OPAQUE } else { DISABLED_ALPHA };
84            let is_selected = *self.selected.borrow();
85            let item_color = if is_selected { bg_color.lighten(0.3) } else { bg_color };
86            let item_color = RGBA::from(item_color.with_alpha(disabled_alpha));
87
88            // Draw shadow
89            let shadow_color = RGBA::from(PieMenuSlot::ItemShadow.color());
90            let shadow_size = item_radius * 2.0 + Pixels::new(2.0);
91            let shadow_rect = bounds_of_size(shadow_size);
92            let shadow_rounded = RoundedRect::from_rect(shadow_rect, corner_radius);
93            snapshot.push_rounded_clip(&shadow_rounded);
94            snapshot.append_color(&shadow_color, &shadow_rect);
95            snapshot.pop();
96
97            // Draw background square
98            let item_rect = bounds_of_size(item_radius * 2.0);
99            let item_rounded = RoundedRect::from_rect(item_rect, corner_radius);
100            snapshot.push_rounded_clip(&item_rounded);
101            snapshot.append_color(&item_color, &item_rect);
102            snapshot.pop();
103
104            // Draw selection ring outline when keyboard-selected
105            if is_selected {
106                let selection_ring_radius = item_radius + Pixels::new(3.0);
107                let selection_ring_rect = bounds_of_size(selection_ring_radius * 2.0);
108                let selection_ring_rounded = RoundedRect::from_rect(selection_ring_rect, corner_radius);
109                let selection_ring_color = RGBA::from(PieMenuSlot::SelectionRing.color());
110                let stroke = Stroke::new(2.0);
111                let builder = PathBuilder::new();
112                builder.add_rounded_rect(&selection_ring_rounded);
113                let path = builder.to_path();
114                snapshot.append_stroke(&path, &stroke, &selection_ring_color);
115            }
116
117            // Draw icon
118            let icon_name = self.icon_name.borrow();
119            let icon_name = icon_name.as_str();
120            if !icon_name.is_empty()
121                && let Some(display) = gdk::Display::default()
122            {
123                let icon_theme = IconTheme::for_display(&display);
124                let icon_size = self
125                    .icon_size
126                    .borrow()
127                    .map(|size| Pixels::new(f64::from(size.as_raw())))
128                    .unwrap_or(item_radius * 0.6);
129                let paintable = icon_theme.lookup_icon(
130                    icon_name,
131                    &[icon_name],
132                    icon_size.as_f64() as i32,
133                    1,
134                    gtk4::TextDirection::None,
135                    gtk4::IconLookupFlags::FORCE_SYMBOLIC,
136                );
137                let icon_pos = center - Vec2::<Pixels>::splat(icon_size / 2.0);
138                snapshot.translate(&Point::from(icon_pos.to_f32()));
139                paintable.snapshot(snapshot, icon_size.as_f64(), icon_size.as_f64());
140                snapshot.translate(&Point::from((-icon_pos).to_f32()));
141            }
142
143            // Draw label below icon
144            let label_text = self.label.borrow();
145            let label_text = label_text.as_str();
146            if !label_text.is_empty() {
147                let pango_context = obj.pango_context();
148                let pango_layout = gtk4::pango::Layout::new(&pango_context);
149                pango_layout.set_text(label_text);
150                let font_desc = FontDescription::from_string("Sans 7");
151                pango_layout.set_font_description(Some(&font_desc));
152
153                let label_color = *self.label_color.borrow();
154                let label_color = RGBA::from(label_color.with_alpha(disabled_alpha));
155                let (_ink_rect, logical_rect) = pango_layout.extents();
156                let label_width = logical_rect.width() as f64 / gtk4::pango::SCALE as f64;
157                let label_pos = Vec2::<Pixels>::new(center.x - label_width / 2.0, center.y + item_radius.as_f64());
158
159                // Label shadow
160                let shadow_offset = Vec2::<Pixels>::splat(Pixels::new(1.0));
161                let shadow_color = RGBA::from(PieMenuSlot::LabelShadow.color());
162                snapshot.translate(&Point::from((label_pos + shadow_offset).to_f32()));
163                snapshot.append_layout(&pango_layout, &shadow_color);
164                snapshot.translate(&Point::from((-(label_pos + shadow_offset)).to_f32()));
165
166                // Label
167                snapshot.translate(&Point::from(label_pos.to_f32()));
168                snapshot.append_layout(&pango_layout, &label_color);
169                snapshot.translate(&Point::from((-label_pos).to_f32()));
170            }
171        }
172
173        fn measure(&self, _orientation: gtk4::Orientation, _for_size: i32) -> (i32, i32, i32, i32) {
174            let item_radius = self.item_radius.borrow().as_raw();
175            let size = (item_radius * 2.0 + 20.0) as i32;
176            (size, size, -1, -1)
177        }
178    }
