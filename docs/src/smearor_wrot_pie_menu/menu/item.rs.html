24/// A single item in a pie menu
25#[derive(Debug, Clone, Serialize, Deserialize, TypedBuilder)]
26pub struct MenuItem {
27    /// The unique identifier
28    #[builder(setter(into))]
29    pub id: String,
30    /// The angle of the menu item in degrees
31    pub angle: f32,
32    /// The radius of the menu item in pixels (optional, uses default if not set)
33    #[builder(default, setter(into, strip_option))]
34    pub radius: Option<f32>,
35    /// The name of the event to be triggered when the menu item is selected
36    #[builder(setter(into))]
37    pub event: String,
38    /// Whether the menu item is enabled (clickable). Defaults to `true`.
39    #[builder(default = true)]
40    pub enabled: bool,
41    /// When `true`, the item's `angle` is treated as a fixed semantic position.
42    /// Auto-distribution will not re-assign this item's angle.
43    /// The remaining items are distributed in the gaps between fixed items.
44    #[builder(default = false)]
45    pub fixed_position: bool,
46    /// Whether the pie menu closes after this item is clicked. Defaults to `true`.
47    #[builder(default = true)]
48    pub close_on_click: bool,
49    /// Optional submenu items. When present, selecting this item
50    /// opens a nested ring instead of sending an event.
51    ///
52    /// Submenu items follow the same `fixed_position` / flexible angle
53    /// distribution rules as top-level items.
54    ///
55    /// **ID uniqueness**: The `id` field must be globally unique across the
56    /// entire menu tree (all levels). This simplifies lookup operations
57    /// (`open_submenu`, `get_submenu_items`, `set_submenu_items`) to a flat
58    /// search instead of a tree traversal from the root via `submenu_stack`.
59    /// Duplicate IDs at any level are undefined behavior.
60    #[builder(default, setter(strip_option))]
61    pub submenu: Option<Vec<MenuItem>>,
62
63    /// The widget type name used to resolve the factory from the
64    /// registry. When `None`, defaults to `"circle"` — preserving
65    /// existing behavior.
66    ///
67    /// This field is serializable so that widget types can be
68    /// stored in JSON/TOML configuration files.
69    #[builder(default, setter(strip_option, into))]
70    #[serde(default)]
71    pub widget_type: Option<String>,
72
73    /// Type-specific widget configuration as a `serde_json::Value`.
74    ///
75    /// When `None`, the factory's `Config::default()` is used.
76    /// When `Some(value)`, the value is deserialized into the factory's
77    /// `Config` type by the `MenuItemWidgetFactoryErased` blanket impl.
78    ///
79    /// This field is serializable so that widget configuration can
80    /// be stored in JSON/TOML configuration files. The schema is
81    /// defined by the factory's `Config` type — type safety is
82    /// provided at the factory level, not at the `MenuItem` level.
83    #[builder(default, setter(strip_option))]
84    #[serde(default)]
85    pub widget_config: Option<serde_json::Value>,
86
87    /// Optional non-square allocation size for widget content.
88    /// When `None`, the item's `radius` is used for a square
89    /// allocation of `2 * radius` pixels. When `Some(ItemSize)`,
90    /// the widget is allocated with the specified dimensions.
91    ///
92    /// This field is serializable.
93    #[builder(default, setter(strip_option))]
94    #[serde(default)]
95    pub content_size: Option<ItemSize>,
96
97    /// Whether the widget rotates with the ring or stays upright.
98    ///
99    /// When `true` (default), the widget rotates with the ring.
100    /// When `false`, the widget stays upright.
101    #[builder(default = true)]
102    #[serde(default = "default_content_rotates")]
103    pub content_rotates: bool,
104}

106/// Manual extension of the generated `MenuItemBuilder` providing a
107/// typed `config()` setter that accepts any `Serialize` type.
108///
109/// This avoids requiring callers to wrap configs in
110/// `serde_json::to_value(&...).unwrap()` manually.
111#[allow(non_camel_case_types)]
112impl<__id, __angle, __radius, __event, __enabled, __fixed_position, __close_on_click, __submenu, __widget_type, __content_size, __content_rotates>
113    MenuItemBuilder<(
114        __id,
115        __angle,
116        __radius,
117        __event,
118        __enabled,
119        __fixed_position,
120        __close_on_click,
121        __submenu,
122        __widget_type,
123        (),
124        __content_size,
125        __content_rotates,
126    )>
127{
128    /// Sets the `widget_config` from any `Serialize` type.
129    ///
130    /// The config is serialized to `serde_json::Value` internally.
131    /// This is the typed equivalent of `.widget_config(serde_json::to_value(&config).unwrap())`.
132    pub fn config<T: Serialize>(
133        self,
134        config: T,
135    ) -> MenuItemBuilder<(
136        __id,
137        __angle,
138        __radius,
139        __event,
140        __enabled,
141        __fixed_position,
142        __close_on_click,
143        __submenu,
144        __widget_type,
145        (Option<serde_json::Value>,),
146        __content_size,
147        __content_rotates,
148    )> {
149        self.widget_config(serde_json::to_value(&config).unwrap_or(serde_json::Value::Null))
150    }
151}

153impl MenuItem {
154    /// Returns the radius, falling back to the default if not set
155    pub fn radius(&self) -> f32 {
156        self.radius.unwrap_or(DEFAULT_MENU_ITEM_RADIUS)
157    }
158
159    /// Recursively searches for an item with the given id in this item's subtree.
160    /// Returns a clone of the found item, or `None` if not found.
161    pub fn find_recursive(&self, id: &str) -> Option<MenuItem> {
162        if self.id == id {
163            return Some(self.clone());
164        }
165        if let Some(submenu) = &self.submenu {
166            for item in submenu {
167                if let Some(found) = item.find_recursive(id) {
168                    return Some(found);
169                }
170            }
171        }
172        None
173    }
174
175    /// Recursively replaces the submenu of the item with `parent_id`.
176    /// Returns `true` if the parent was found and updated.
177    pub fn replace_submenu_recursive(&mut self, parent_id: &str, new_submenu: &[MenuItem]) -> bool {
178        if self.id == parent_id {
179            self.submenu = Some(new_submenu.to_vec());
180            return true;
181        }
182        if let Some(submenu) = &mut self.submenu {
183            for item in submenu.iter_mut() {
184                if item.replace_submenu_recursive(parent_id, new_submenu) {
185                    return true;
186                }
187            }
188        }
189        false
190    }
191}
