8/// Type-erased factory trait for registry storage.
9///
10/// This trait allows the registry to store factories with different
11/// `Config` types in a single `HashMap<String, Box<dyn MenuItemWidgetFactoryErased>>`.
12///
13/// A blanket implementation automatically converts any
14/// `MenuItemWidgetFactory` into a `MenuItemWidgetFactoryErased` by
15/// deserializing `item.widget_config` into the factory's `Config` type.
16pub trait MenuItemWidgetFactoryErased {
17    /// Returns the unique type name for this factory.
18    fn type_name(&self) -> &str;
19
20    /// Builds and returns a GTK4 widget, deserializing the config automatically.
21    fn build(&self, item: &MenuItem, context: &MenuItemContext) -> Widget;
22}
23
24impl<F> MenuItemWidgetFactoryErased for F
25where
26    F: MenuItemWidgetFactory,
27    F::Config: Serialize + DeserializeOwned + Default,
28{
29    fn type_name(&self) -> &str {
30        <Self as MenuItemWidgetFactory>::type_name(self)
31    }
32
33    fn build(&self, item: &MenuItem, context: &MenuItemContext) -> Widget {
34        let config: F::Config = item
35            .widget_config
36            .as_ref()
37            .and_then(|value| serde_json::from_value(value.clone()).ok())
38            .unwrap_or_default();
39        <Self as MenuItemWidgetFactory>::build(self, item, &config, context)
40    }
41}

44mod tests {
45    use super::*;
46    use crate::menu::widget_factory::MenuItemWidgetFactory;
47    use gtk4::prelude::Cast;
48    use serde::Deserialize;

50    struct DummyFactory;
51
52    #[derive(Debug, Clone, Serialize, Deserialize, Default, PartialEq)]
53    struct DummyConfig {
54        value: f64,
55    }
56
57    impl MenuItemWidgetFactory for DummyFactory {
58        type Config = DummyConfig;
59
60        fn type_name(&self) -> &str {
61            "dummy"
62        }
63
64        fn build(&self, _item: &MenuItem, config: &DummyConfig, _context: &MenuItemContext) -> Widget {
65            let label = gtk4::Label::new(Some(&format!("{}", config.value)));
66            label.upcast::<Widget>()
67        }
68    }
69
70    #[test]
71    fn test_erased_type_name() {
72        let factory = DummyFactory;
73        let erased: &dyn MenuItemWidgetFactoryErased = &factory;
74        assert_eq!(erased.type_name(), "dummy");
75    }

77    #[test]
78    #[ignore = "requires GTK display environment"]
79    fn test_erased_build_deserializes_config() {
80        crate::test_util::ensure_gtk_init();
81        let factory = DummyFactory;
82        let item = MenuItem::builder()
83            .id("test")
84            .angle(0.0)
85            .event("event")
86            .widget_config(serde_json::json!({ "value": 99.0 }))
87            .build();
88        let context = MenuItemContext {
89            id: "test".to_string(),
90            event: "event".to_string(),
91            trigger_event: std::boxed::Box::new(|| {}),
92        };
93        let erased: &dyn MenuItemWidgetFactoryErased = &factory;
94        let widget = erased.build(&item, &context);
95        let label = widget.downcast_ref::<gtk4::Label>().expect("expected Label");
96        assert_eq!(label.text(), "99");
97    }
