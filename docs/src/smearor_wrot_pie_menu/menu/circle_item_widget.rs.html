61    impl WidgetImpl for CircleItemWidgetImpl {
62        fn snapshot(&self, snapshot: &gtk4::Snapshot) {
63            let obj = self.obj();
64            let center = obj.center();
65            let item_radius = Pixels::new(f64::from(self.item_radius.borrow().as_raw()));
66
67            // Bounding square of a centered circle of the given radius -
68            // converts to graphene coordinates at the draw boundary.
69            let bounds_at = |radius: Pixels<f64>| graphene::Rect::from(Rect::<Pixels>::centered(center, Vec2::<Pixels>::splat(radius * 2.0)).to_f32());
70
71            let bg_color = *self.bg_color.borrow();
72            let disabled_alpha = if *self.enabled.borrow() { Alpha::OPAQUE } else { DISABLED_ALPHA };
73            let is_selected = *self.selected.borrow();
74            let item_color = if is_selected { bg_color.lighten(0.3) } else { bg_color };
75            let item_color = gdk::RGBA::from(item_color.with_alpha(disabled_alpha));
76
77            // Draw shadow
78            let shadow_color = gdk::RGBA::from(PieMenuSlot::ItemShadow.color());
79            let shadow_radius = item_radius + Pixels::new(2.0);
80            let shadow_rect = bounds_at(shadow_radius);
81            let shadow_rounded = gsk::RoundedRect::from_rect(shadow_rect, shadow_radius.as_f64() as f32);
82            snapshot.push_rounded_clip(&shadow_rounded);
83            snapshot.append_color(&shadow_color, &shadow_rect);
84            snapshot.pop();
85
86            // Draw background circle
87            let item_rect = bounds_at(item_radius);
88            let item_rounded = gsk::RoundedRect::from_rect(item_rect, item_radius.as_f64() as f32);
89            snapshot.push_rounded_clip(&item_rounded);
90            snapshot.append_color(&item_color, &item_rect);
91            snapshot.pop();
92
93            // Draw selection ring outline when keyboard-selected
94            if is_selected {
95                let selection_ring_radius = item_radius + Pixels::new(3.0);
96                let selection_ring_rect = bounds_at(selection_ring_radius);
97                let selection_ring_rounded = gsk::RoundedRect::from_rect(selection_ring_rect, selection_ring_radius.as_f64() as f32);
98                let selection_ring_color = gdk::RGBA::from(PieMenuSlot::SelectionRing.color());
99                let stroke = gsk::Stroke::new(2.0);
100                let builder = gsk::PathBuilder::new();
101                builder.add_rounded_rect(&selection_ring_rounded);
102                let path = builder.to_path();
103                snapshot.append_stroke(&path, &stroke, &selection_ring_color);
104            }
105
106            // Draw icon
107            let icon_name = self.icon_name.borrow();
108            let icon_name = icon_name.as_str();
109            if !icon_name.is_empty()
110                && let Some(display) = gdk::Display::default()
111            {
112                let icon_theme = gtk4::IconTheme::for_display(&display);
113                let icon_size = (item_radius * 0.6).as_f64() as i32;
114                let paintable = icon_theme.lookup_icon(icon_name, &[icon_name], icon_size, 1, gtk4::TextDirection::None, gtk4::IconLookupFlags::FORCE_SYMBOLIC);
115                let icon_pos = center - Vec2::<Pixels>::splat(Pixels::new(f64::from(icon_size) / 2.0));
116                snapshot.translate(&graphene::Point::from(icon_pos.to_f32()));
117                paintable.snapshot(snapshot, icon_size as f64, icon_size as f64);
118                snapshot.translate(&graphene::Point::from((-icon_pos).to_f32()));
119            }
120
121            // Draw label below icon
122            let label_text = self.label.borrow();
123            let label_text = label_text.as_str();
124            if !label_text.is_empty() {
125                let pango_context = obj.pango_context();
126                let pango_layout = gtk4::pango::Layout::new(&pango_context);
127                pango_layout.set_text(label_text);
128                let font_desc = gtk4::pango::FontDescription::from_string("Sans 7");
129                pango_layout.set_font_description(Some(&font_desc));
130
131                let label_color = *self.label_color.borrow();
132                let label_color = gdk::RGBA::from(label_color.with_alpha(disabled_alpha));
133                let (_ink_rect, logical_rect) = pango_layout.extents();
134                let label_width = logical_rect.width() as f64 / gtk4::pango::SCALE as f64;
135                let label_pos = Vec2::<Pixels>::new(center.x - label_width / 2.0, center.y + item_radius.as_f64());
136
137                // Label shadow
138                let shadow_offset = Vec2::<Pixels>::splat(Pixels::new(1.0));
139                let shadow_color = gdk::RGBA::from(PieMenuSlot::LabelShadow.color());
140                snapshot.translate(&graphene::Point::from((label_pos + shadow_offset).to_f32()));
141                snapshot.append_layout(&pango_layout, &shadow_color);
142                snapshot.translate(&graphene::Point::from((-(label_pos + shadow_offset)).to_f32()));
143
144                // Label
145                snapshot.translate(&graphene::Point::from(label_pos.to_f32()));
146                snapshot.append_layout(&pango_layout, &label_color);
147                snapshot.translate(&graphene::Point::from((-label_pos).to_f32()));
148            }
149        }
150
151        fn measure(&self, _orientation: gtk4::Orientation, _for_size: i32) -> (i32, i32, i32, i32) {
152            let item_radius = self.item_radius.borrow().as_raw();
153            let size = (item_radius * 2.0 + 20.0) as i32;
154            (size, size, -1, -1)
155        }
156    }
